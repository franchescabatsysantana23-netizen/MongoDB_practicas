use('ecommerce');

const pipeline =[
        {

         $lookup:{
           from: "users",
           localField: "user_id",
           foreignField: "_id",
           as: "detalleUsuario"
        },
        {
         $lookup:{
           from: 'branches',
           localField: 'branch_id',
           foreignField: '_id',
           as: 'detalleSucursal', 
        },
        {
          $project:{
          _id:1,
          fecha:1,
          productos:1,
          detalleUsuario:1,
          user: {$arrayElemenAt: ['$detalleUsuario', 0]},
          branch: {
              _id: {$arrayElemAt:['$detalleSucursal',0]},
              nombre: {$arrayElemAt: ['$detalleSucursal.nombre',0]},
              province: {$arrayElemAt: ['$detalleSucursal.province',0]}, 
                  },

         {
          $unwind: '$productos'
         }
           
           }
        }
];
const resultado = db.pedidos.aggregate(pipeline).toArray();
print(resultado)
